POST /api/v1/book/list

获取书籍列表 API 实现方案

用户聚焦天体后，面板展示该天体下允许公开的书籍。现有控制器对 ID 为 2 的天体返回空数组，对其他合法 ID 返回固定八本书。本稿拟用数据库读取替换这两个分支，并让真实书籍 ID 能从书单一路传到目录请求。

接口契约 · 查询与状态码 · 审阅 Go 代码 · 前端改动 · 验收

请求与响应

现有路由接收公开的 JSON POST 请求。沿用 param/req/v1/book.ListReq，其 planet.id 已按带引号的十进制字符串解码并校验为正 int64。接口一次返回该天体的完整公开书单，无分页参数。

请求示例

{"planet":{"id":"9007199254740993"}}

候选成功响应

{
  "code": 200,
  "message": "success",
  "data": {
    "bookList": [
      {"id": "9007199254740995", "title": "操作系统入门"},
      {"id": "9007199254740997", "title": "进程与调度"}
    ]
  }
}

每项只含书籍主键 id 和数据库原文 title。后台创建书籍时使用雪花 ID。按项目的大整数约定，拟将 bookList[].id 从当前文档中的 JSON 数字改为十进制字符串，同时将目录请求的 book.id 改为相同类型。数据库和业务层继续使用 int64。

哪些书可以返回

参考后台 OptionsByPlanetID 的读法，在同一个 MySQL 只读、可重复读事务中执行两次查询。第一条确定天体是否存在，第二条按该天体的配置读取可见书籍。两次读共享快照，即使管理端同时调整可见性或书序，一次响应也使用同一版本的数据。

SELECT id FROM planet WHERE id = ? LIMIT 1;

SELECT id, title FROM book
WHERE planet_id = ? AND display_status = 'visible'
ORDER BY sort ASC;

按管理员维护的 book.sort 字段升序返回可见书籍。查询只依赖 planet 和 book，一本可见书即使还没有目录或文章，也会出现在书单。

待评审的空结果与错误语义

现行前台契约只列出成功响应。建议沿用后台的 20102 表示请求引用的天体已不存在，并保留前台既有的 HTTP 200 和业务码包装。这样，面板能区分正常空书单和失败重试。

情形业务码载荷和前端表现
存在可见书籍200data.bookList 为有序数组，展示书名和首本快捷入口。
天体存在，无书或全部隐藏200{"bookList":[]}，面板列表为空，首本入口隐藏。
JSON 不合法，或天体 ID 缺失、为数字、非正、溢出10101data: {}，沿用已有入参校验。
格式合法，但天体不存在20102data: {}，公开消息为“所选天体不存在,请刷新天体列表”。
查询或事务提交失败10201data: {}，复用 DBError()，面板可重试。
其他内部错误500data: {}，复用 InternalError()。

controller/v1/book/book.go

替换现有控制器。只向调用方公开书籍 ID 和标题，提前初始化切片以保证空书单序列化为数组。

package book

import (
	"errors"
	"net/http"
	"strconv"

	"SolarNotesAPI/biz"
	reqBook "SolarNotesAPI/param/req/v1/book"
	respBook "SolarNotesAPI/param/resp/v1/book"
	"SolarNotesAPI/resp_body"
	"SolarNotesAPI/sys_error"

	"github.com/gin-gonic/gin"
)

// List 返回所选天体的公开书单.
func List(c *gin.Context) {
	var req reqBook.ListReq
	var body resp_body.RespBody
	if err := c.ShouldBindJSON(&req); err != nil {
		body.ValidateFailed(err)
		c.JSON(http.StatusOK, body)
		return
	}

	var service biz.Book
	books, err := service.List(c.Request.Context(), req.Planet.ID)
	if err != nil {
		var dbErr *sys_error.DatabaseError
		switch {
		case errors.Is(err, sys_error.PlanetNotFoundErr):
			body.PlanetNotFoundError()
		case errors.As(err, &dbErr):
			body.DBError()
		default:
			body.InternalError()
		}
		c.JSON(http.StatusOK, body)
		return
	}

	items := make([]respBook.ListItem, 0, len(books))
	for _, book := range books {
		items = append(items, respBook.ListItem{
			ID:    strconv.FormatInt(book.ID, 10),
			Title: book.Title,
		})
	}
	body.Success(map[string]any{"bookList": items})
	c.JSON(http.StatusOK, body)
}

biz/book.go

新增业务实体。天体不存在时返回业务错误，其他 DAO 故障用现有数据库错误包装。

package biz

import (
	"context"
	"errors"

	"SolarNotesAPI/model"
	"SolarNotesAPI/sys_error"
)

type Book struct {
	ID    int64
	Title string
}

// List 保留 DAO 提供的展示顺序, 将数据访问错误转换为接口可识别的类别.
func (Book) List(ctx context.Context, planetID int64) ([]Book, error) {
	if ctx == nil || planetID <= 0 {
		return nil, &sys_error.InternalError{Cause: errors.New("biz.Book.List: invalid context or planet ID")}
	}

	var dao model.Book
	rows, err := dao.ListVisibleByPlanet(ctx, planetID)
	if errors.Is(err, model.PlanetNotFoundErr) {
		return nil, sys_error.PlanetNotFoundErr
	}
	if err != nil {
		return nil, &sys_error.DatabaseError{Cause: err}
	}

	books := make([]Book, 0, len(rows))
	for _, row := range rows {
		books = append(books, Book{ID: row.ID, Title: row.Title})
	}
	return books, nil
}

model/book.go

新增 DAO。GORM 通过现有单数表名配置推导表名，每次查询使用同一个事务句柄。

package model

import (
	"context"
	"database/sql"
	"errors"

	"SolarNotesAPI/db"

	"gorm.io/gorm"
)

var PlanetNotFoundErr = errors.New("planet not found")

// Book 映射书单查询需要返回的两列.
type Book struct {
	ID    int64  `gorm:"column:id;primaryKey;autoIncrement:false"`
	Title string `gorm:"column:title"`
}

// ListVisibleByPlanet 在同一只读快照内检查天体并查询书单.
func (Book) ListVisibleByPlanet(ctx context.Context, planetID int64) ([]Book, error) {
	if db.DB == nil {
		return nil, errors.New("database connection is not initialized")
	}

	var books []Book
	opts := &sql.TxOptions{Isolation: sql.LevelRepeatableRead, ReadOnly: true}
	err := db.DB.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
		var planet Planet
		err := tx.Select("id").Where("id = ?", planetID).Take(&planet).Error
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return PlanetNotFoundErr
		}
		if err != nil {
			return err
		}

		return tx.Select("id", "title").
			Where("planet_id = ? AND display_status = ?", planetID, "visible").
			Order("sort ASC").
			Find(&books).Error
	}, opts)
	if err != nil {
		return nil, err
	}
	return books, nil
}

验收与实库核对

评审前验证。从本页提取的七份 Go 代码在临时工程副本中通过了构建和 go vet。SQLite 内存库与 httptest 覆盖了可见性、跨天体隔离、排序、空书单、大 ID 传入目录，以及三类失败响应。在前端临时加载候选改动后，首本入口、多书列表第二项和直接深链都将原始字符串传入目录请求，包括 "9223372036854775807"。生产用 MySQL 的事务行为与真实数据留到正式实施时验证。

  1. 隔离与排序。在测试数据中为两个天体放入书籍，混合可见与隐藏状态，并让主键顺序与书序不同。结果只含目标天体的可见记录，按 sort 从小到大排列，书名逐字一致。
  2. 空表与失败。分别验证存在但无书、全部隐藏、天体不存在、查询错误，以及旧 mock 特判的天体 ID "2"。ID 为 2 时同样只由实际数据决定结果。
  3. ID 精度。用超过 Number.MAX_SAFE_INTEGER 的书籍 ID 检查首本入口、第二本链接和直接打开阅读页。书单响应、路由和目录请求中的 ID 必须逐位相同；裸数字、负数、零及超出正 int64 的 ID 应在入参绑定时失败。
  4. 实施后实库核对。从开发环境天体列表选择真实 ID，按下列方式请求书单。用独立只读 SQL 筛选该天体的可见书籍，按 sort ASC 核对返回顺序，并校验行数、书名和每个书籍 ID 的十进制原文，再用其中一本书的 ID 请求目录。
